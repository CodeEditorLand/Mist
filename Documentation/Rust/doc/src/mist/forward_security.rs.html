1//! DNS Forward Security Module
2//!
3//! This module provides security wrappers for DNS forwarding, implementing
4//! an allowlist-based approach to prevent sidecars from reaching arbitrary
5//! external hosts via DNS.
6//!
7//! ## SECURITY ARCHITECTURE
8//!
9//! The RestrictedForwardAuthority implements a defense-in-depth measure:
10//!
11//! ```text
12//! Query ──► Is *.editor.land? ──► Authoritative (Local)
13//!            │ No
14//!            ▼
15//!     Is in Allowlist? ──► Forward to Upstream
16//!            │ No
17//!            ▼
18//!        Return REFUSED
19//! ```
20//!
21//! This prevents sidecars from using DNS to reach unapproved external
22//! services, limiting the attack surface for compromised processes.
23//!
24//! # Note
25//!
26//! This module is currently a simplified stub. The Authority trait implementations
27//! have been removed because Hickory DNS 0.25.x has significant API changes that
28//! require complex lifetime handling. The main DNS functionality uses
29//! `InMemoryAuthority` from hickory-server which implements Authority correctly.
30
31use anyhow::{Result, anyhow};
32use hickory_proto::rr::Name;
33
34/// Creates a default forward authority allowlist.
35///
36/// Returns an iterator of domain names that are allowed to be forwarded.
37///
38/// # Default Allowlist
39///
40/// - `update.editor.land.` - For application updates
41///
42/// # Example
43///
44/// ```rust
45/// use Mist::forward_security::default_forward_allowlist;
46///
47/// let allowlist:Vec<Name> = default_forward_allowlist().map(|n| n.unwrap()).collect();
48/// ```
49pub fn default_forward_allowlist() -> impl Iterator<Item = Result<Name>> {
50	vec![
51		Name::from_ascii("update.editor.land."),
52	]
53	.into_iter()
54	.map(|r| r.map_err(|e| anyhow!("Failed to parse domain name: {}", e)))
55}

57#[cfg(test)]
58mod tests {
59	use super::*;
60
61	#[test]
62	fn test_allowlist_generation() {
63		let allowlist: Vec<Name> = default_forward_allowlist()
64			.filter_map(|r| r.ok())
65			.collect();
66		
67		assert!(!allowlist.is_empty(), "Allowlist should not be empty");
68		assert_eq!(allowlist.len(), 2, "Should have 2 domains in allowlist");
69	}
