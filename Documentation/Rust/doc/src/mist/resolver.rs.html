1//! DNS Resolver Module
2//!
3//! This module provides DNS resolution capabilities for the CodeEditorLand

5//! Hickory DNS server for editor.land domains.

39/// Custom DNS resolver that uses the local Hickory DNS server.
40///
41/// This resolver is designed for use with reqwest's DNS override functionality.
42/// It wraps a TokioResolver and implements the `reqwest::dns::Resolve` trait.
43///
44/// # Security
45///
46/// This resolver ensures that all DNS queries go through the local DNS server,
47/// which resolves `*.editor.land` domains to `127.x.x.x` addresses as a
48/// defense-in-depth measure.
49///
50/// # Note
51///
52/// This is a stub implementation for compilation. The actual resolver would
53/// connect to the local DNS server and resolve queries.
54pub struct LandDnsResolver;
55

72// Implement the reqwest::dns::Resolve trait
73// This allows the resolver to be used with reqwest's DNS override
74impl reqwest::dns::Resolve for LandDnsResolver {
75	fn resolve(&self, name: reqwest::dns::Name) -> reqwest::dns::Resolving {
76		let name_str = name.as_str().to_string();
77
78		Box::pin(async move {
79			// SECURITY: Defense-in-depth IP validation for *.editor.land domains
80			//
81			// This check ensures that any DNS response for *.editor.land domains
82			// only returns 127.x.x.x addresses. This provides protection against:
83			//
84			// 1. **Upstream DNS poisoning**: If an attacker compromises the upstream DNS or
85			//    the forward authority, they cannot return external IPs for editor.land
86			//    domains
87			//
88			// 2. **Zone file corruption**: If the zone file is modified to point to
89			//    external IPs, this validation prevents traffic from leaving localhost
90			//
91			// 3. **Process compromise**: If the DNS server process is compromised, this
92			//    prevents it from redirecting editor.land traffic to external hosts
93			//
94			// Any non-editor.land domain is allowed to resolve to any IP (subject to
95			// the forward allowlist restrictions in RestrictedForwardAuthority).
96			let is_editor_land = name_str.ends_with(".editor.land") || name_str == "editor.land";
97
98			if is_editor_land {
99				// For editor.land domains, return loopback addresses
100				let addrs = vec![
101					SocketAddr::new(IpAddr::V4(Ipv4Addr::new(127, 0, 0, 1)), 0),
102				];
103				Ok(Box::new(addrs.into_iter()) as Box<dyn Iterator<Item = SocketAddr> + Send>)
104			} else {
105				// For non-editor.land domains, return empty (will fall back to system DNS)
106				// In production, this would forward to the upstream DNS server
107				Ok(Box::new(std::iter::empty()) as Box<dyn Iterator<Item = SocketAddr> + Send>)
108			}
109		})
110	}
111}
